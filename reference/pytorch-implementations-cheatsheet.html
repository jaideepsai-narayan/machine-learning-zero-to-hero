1. The Artificial Neuron (Perceptron) & Training Loop

A single neuron taking inputs $x$, multiplying by weights $w$, adding bias $b$, and passing through a Sigmoid activation:

import torch
import torch.nn as nn
import torch.optim as optim

# 1. Define Single Neuron / Perceptron Model
class Perceptron(nn.Module):
    def __init__(self, input_dim):
        super().__init__()
        self.linear = nn.Linear(input_dim, 1)  # z = w*x + b
        self.sigmoid = nn.Sigmoid()            # a = 1 / (1 + e^-z)

    def forward(self, x):
        return self.sigmoid(self.linear(x))

# 2. Instantiate Model, Loss & Optimizer
model = Perceptron(input_dim=4)
criterion = nn.BCELoss()  # Binary Cross-Entropy Loss
optimizer = optim.SGD(model.parameters(), lr=0.01)

# 3. Standard PyTorch Training Step
inputs = torch.randn(10, 4)
labels = torch.randint(0, 2, (10, 1)).float()

# Forward pass
predictions = model(inputs)
loss = criterion(predictions, labels)

# Backward pass (Backpropagation!)
optimizer.zero_grad()
loss.backward()  # Computes dLoss/dw and dLoss/db automatically
optimizer.step() # Updates weights: w = w - lr * grad

2. Multi-Layer Perceptron (MLP / Deep Neural Network)

Stacked hidden layers with non-linear activation functions (ReLU):

class MultiLayerPerceptron(nn.Module):
    def __init__(self, input_dim, hidden_dim, num_classes):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
            nn.Dropout(0.2), # Regularization
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, num_classes)
        )

    def forward(self, x):
        return self.net(x)

# Training setup for multi-class classification
model = MultiLayerPerceptron(input_dim=784, hidden_dim=128, num_classes=10)
criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(model.parameters(), lr=0.001)

3. Convolutional Neural Network (CNN) for Computer Vision

Convolution layers, ReLU activations, and Max Pooling for spatial image classification:

class ConvNet(nn.Module):
    def __init__(self, num_classes=10):
        super().__init__()
        self.features = nn.Sequential(
            # Block 1: Conv -> ReLU -> MaxPool
            nn.Conv2d(in_channels=3, out_channels=16, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(kernel_size=2, stroke=2),
            
            # Block 2
            nn.Conv2d(in_channels=16, out_channels=32, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(kernel_size=2, stroke=2)
        )
        self.classifier = nn.Linear(32 * 7 * 7, num_classes)

    def forward(self, x):
        x = self.features(x)
        x = torch.flatten(x, start_dim=1) # Flatten spatial feature map
        return self.classifier(x)

4. Recurrent Neural Network (LSTM) for Sequential Data & NLP

Long Short-Term Memory network for processing text sequences and time series:

class TextLSTM(nn.Module):
    def __init__(self, vocab_size, embed_dim, hidden_dim, num_classes):
        super().__init__()
        self.embedding = nn.Embedding(vocab_size, embed_dim)
        self.lstm = nn.LSTM(embed_dim, hidden_dim, batch_first=True)
        self.fc = nn.Linear(hidden_dim, num_classes)

    def forward(self, x):
        embedded = self.embedding(x)          # [batch, seq_len, embed_dim]
        _, (h_n, _) = self.lstm(embedded)      # h_n shape: [1, batch, hidden_dim]
        return self.fc(h_n[-1])

5. Transformer Block with Multi-Head Self-Attention

Parallel sequence self-attention block ($Q, K, V$ dot products):

class TransformerBlock(nn.Module):
    def __init__(self, embed_dim, num_heads):
        super().__init__()
        self.attn = nn.MultiheadAttention(embed_dim, num_heads, batch_first=True)
        self.norm1 = nn.LayerNorm(embed_dim)
        self.ffn = nn.Sequential(
            nn.Linear(embed_dim, embed_dim * 4),
            nn.ReLU(),
            nn.Linear(embed_dim * 4, embed_dim)
        )
        self.norm2 = nn.LayerNorm(embed_dim)

    def forward(self, x):
        # Scaled Dot-Product Self-Attention
        attn_out, _ = self.attn(query=x, key=x, value=x)
        x = self.norm1(x + attn_out) # Residual Skip Connection + LayerNorm
        ffn_out = self.ffn(x)
        return self.norm2(x + ffn_out)
